# Plays ~19 seconds of 2 Player Arcade. Run with
#   --quick-arcade=2:1:normal --record-input=tests/sync/2p-arcade.otr
# The game starts the level by itself; only the gameplay keys are scripted.
window 640 400
wait 3000

down Space
down Left
wait 1500
up Left
down Up
wait 1000
up Up
down Right
wait 3000
up Right
key Left_Ctrl 300
down Down
wait 800
up Down
wait 3000
down Left
wait 2000
up Left
key Return 200
down Right
wait 2500
up Right
up Space
wait 2000
quit
